set(IMU_SOURCES imu.c vendor/Fusion/Fusion/FusionAhrs.c)
set(IMU_INCLUDES include vendor/Fusion/Fusion)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${IMU_SOURCES} INCLUDE_DIRS ${IMU_INCLUDES})
    target_compile_definitions(${COMPONENT_LIB} PUBLIC FUSION_USE_NORMAL_SQRT)
else()
    cmake_minimum_required(VERSION 3.16)
    project(ainekio_imu C)
    enable_testing()
    add_library(ainekio_imu STATIC ${IMU_SOURCES})
    target_include_directories(ainekio_imu PUBLIC ${IMU_INCLUDES})
    target_compile_features(ainekio_imu PUBLIC c_std_11)
    target_compile_definitions(ainekio_imu PUBLIC FUSION_USE_NORMAL_SQRT)
    target_compile_options(ainekio_imu PRIVATE -Wall -Wextra -Werror -Wpedantic)
    target_link_libraries(ainekio_imu PUBLIC m)
    add_executable(test_imu tests/test_imu.c)
    target_compile_options(test_imu PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(test_imu PRIVATE ainekio_imu)
    add_test(NAME imu_estimation COMMAND test_imu)
endif()
